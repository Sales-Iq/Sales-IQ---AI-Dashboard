rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isAdminProfile() {
      return signedIn()
        && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    function adminData() {
      return get(/databases/$(database)/documents/admins/$(request.auth.uid)).data;
    }

    function isActiveAdmin() {
      return isAdminProfile()
        && adminData().status == 'active'
        && adminData().role == 'Admin';
    }

    function canCreateAdminProfile() {
      return signedIn()
        && request.resource.data.status == 'active'
        && request.resource.data.role == 'Admin';
    }

    function isStaffProfile() {
      return signedIn()
        && exists(/databases/$(database)/documents/staff/$(request.auth.uid));
    }

    function staffData() {
      return get(/databases/$(database)/documents/staff/$(request.auth.uid)).data;
    }

    function isActiveStaff() {
      return isStaffProfile()
        && staffData().status == 'active'
        && staffData().role == 'Sales Staff';
    }

    function isOwnerOfData(adminId) {
      return signedIn() && adminId == request.auth.uid;
    }

    function activeUser() {
      return isActiveAdmin() || isActiveStaff();
    }

    function isStaffForAdmin(adminId) {
      return isStaffProfile()
        && staffData().status == 'active'
        && staffData().role == 'Sales Staff'
        && staffData().adminId == adminId;
    }

    function stockOnlyUpdate() {
      return request.resource.data
        .diff(resource.data)
        .affectedKeys()
        .hasOnly(['stock', 'status', 'updatedAt']);
    }

    function batchStockOnlyUpdate() {
      return request.resource.data
        .diff(resource.data)
        .affectedKeys()
        .hasOnly(['remainingQuantity', 'status', 'updatedAt']);
    }

    function isSelfRegistration() {
      return signedIn()
        && request.auth.uid == request.resource.data.id
        && request.resource.data.role == 'Sales Staff'
        && request.resource.data.status == 'pending_assignment'
        && request.resource.data.assignedAdminId == null
        && request.resource.data.username != null
        && request.resource.data.username != '';
    }

    function usernameUnique() {
      return !exists(/databases/$(database)/documents/staff/$(request.resource.data.username));
    }

    function staffOnlyProfileUpdate() {
      return request.resource.data
        .diff(resource.data)
        .affectedKeys()
        .hasOnly(['name', 'photoURL', 'lastLoginAt']);
    }

    match /admins/{uid} {
      allow get: if (signedIn() && uid == request.auth.uid) || isActiveAdmin();
      allow list: if isActiveAdmin();
      allow create: if isActiveAdmin() || canCreateAdminProfile();
      allow update: if isActiveAdmin() || (signedIn() && uid == request.auth.uid && request.resource.data.role == resource.data.role && request.resource.data.status == resource.data.status);
      allow delete: if isActiveAdmin();
    }

    match /staff/{uid} {
      allow get: if (signedIn() && uid == request.auth.uid) || isActiveAdmin();
      allow list: if isActiveAdmin();
      allow create: if isActiveAdmin() || (isSelfRegistration() && usernameUnique());
      allow update: if isActiveAdmin() || (signedIn() && uid == request.auth.uid && staffOnlyProfileUpdate());
      allow delete: if isActiveAdmin() || (signedIn() && uid == request.auth.uid);
    }

    match /products/{id} {
      allow read: if isOwnerOfData(resource.data.adminId) || (isActiveAdmin() && resource.data.adminId == request.auth.uid) || (isStaffForAdmin(resource.data.adminId));
      allow create: if isActiveAdmin() && request.resource.data.adminId == request.auth.uid;
      allow update: if (isActiveAdmin() && resource.data.adminId == request.auth.uid) 
        || (isActiveAdmin() && (resource.data.adminId == null || resource.data.adminId == ''))
        || (isStaffForAdmin(resource.data.adminId) && stockOnlyUpdate());
      allow delete: if isActiveAdmin() && resource.data.adminId == request.auth.uid;
    }

    match /productBatches/{id} {
      allow read: if isOwnerOfData(resource.data.adminId) || (isActiveAdmin() && resource.data.adminId == request.auth.uid) || (isStaffForAdmin(resource.data.adminId));
      allow create: if isActiveAdmin() && request.resource.data.adminId == request.auth.uid;
      allow update: if (isActiveAdmin() && resource.data.adminId == request.auth.uid)
        || (isActiveAdmin() && (resource.data.adminId == null || resource.data.adminId == ''))
        || (isStaffForAdmin(resource.data.adminId) && batchStockOnlyUpdate());
      allow delete: if isActiveAdmin() && resource.data.adminId == request.auth.uid;
    }

    match /sales/{id} {
      allow read: if isOwnerOfData(resource.data.adminId) || (isActiveAdmin() && resource.data.adminId == request.auth.uid) || (isStaffForAdmin(resource.data.adminId));
      allow create: if (isActiveAdmin() && request.resource.data.adminId == request.auth.uid) || (isStaffForAdmin(request.resource.data.adminId));
      allow update, delete: if isActiveAdmin() && resource.data.adminId == request.auth.uid
        || (isActiveAdmin() && (resource.data.adminId == null || resource.data.adminId == ''));
    }

    match /customers/{id} {
      allow read: if isOwnerOfData(resource.data.adminId) || (isActiveAdmin() && resource.data.adminId == request.auth.uid) || (isStaffForAdmin(resource.data.adminId));
      allow create: if (isActiveAdmin() && request.resource.data.adminId == request.auth.uid) || (isStaffForAdmin(request.resource.data.adminId));
      allow update, delete: if isActiveAdmin() && resource.data.adminId == request.auth.uid
        || (isActiveAdmin() && (resource.data.adminId == null || resource.data.adminId == ''));
    }

    match /suppliers/{id} {
      allow read: if isOwnerOfData(resource.data.adminId) || isActiveAdmin();
      allow create: if isActiveAdmin() && request.resource.data.adminId == request.auth.uid;
      allow update, delete: if isActiveAdmin() && resource.data.adminId == request.auth.uid;
    }

    match /purchases/{id} {
      allow read: if isOwnerOfData(resource.data.adminId) || isActiveAdmin();
      allow create: if isActiveAdmin() && request.resource.data.adminId == request.auth.uid;
      allow update, delete: if isActiveAdmin() && resource.data.adminId == request.auth.uid;
    }

    match /disposals/{id} {
      allow read: if isOwnerOfData(resource.data.adminId) || (isActiveAdmin() && resource.data.adminId == request.auth.uid) || (isStaffForAdmin(resource.data.adminId));
      allow create: if isActiveAdmin() && request.resource.data.adminId == request.auth.uid;
      allow update, delete: if false;
    }

    match /notifications/{id} {
      allow read: if isOwnerOfData(resource.data.adminId) || (isActiveAdmin() && resource.data.adminId == request.auth.uid) || (isStaffForAdmin(resource.data.adminId));
      allow create: if isActiveAdmin() && request.resource.data.adminId == request.auth.uid;
      allow update, delete: if isActiveAdmin() && resource.data.adminId == request.auth.uid;
    }

    match /forecasting/{id} {
      allow read: if isOwnerOfData(resource.data.adminId) || isActiveAdmin();
      allow create: if isActiveAdmin() && request.resource.data.adminId == request.auth.uid;
      allow update, delete: if isActiveAdmin() && resource.data.adminId == request.auth.uid;
    }

    match /businessSettings/{id} {
      allow read: if isOwnerOfData(resource.data.adminId) || (isActiveAdmin() && resource.data.adminId == request.auth.uid) || (isStaffForAdmin(resource.data.adminId));
      allow create: if isActiveAdmin() && request.resource.data.adminId == request.auth.uid;
      allow update, delete: if isActiveAdmin() && resource.data.adminId == request.auth.uid;
    }
  }
}
